var qi=0,score=0,locked=false,tm=null,deadline=0;
function shuffle(a){a=a.slice();for(var i=a.length-1;i>0;i--){var j=Math.floor(Math.random()*(i+1)),t=a[i];a[i]=a[j];a[j]=t}return a}

  $('qno').textContent=d.boss?'BOSS 👑 ข้อ 5/5':'ข้อ '+(qi+1)+'/5';

  var bar=$('qbar'),qt=$('qtime');

    if(left<=0){clearInterval(tm);qt.textContent='0';timeUp();}
    else{qt.textContent=left;qt.classList.toggle('low',left<=3);}
  },100);
}
function stopTimer(){
  clearInterval(tm);
  var bar=$('qbar');bar.style.transition='none';
  bar.style.width=Math.max(0,(deadline-Date.now())/100)+'%';
  $('qtime').classList.remove('low');
}
